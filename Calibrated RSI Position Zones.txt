// It's inspired by @AdaptiveRSI article: https://x.com/AdaptiveRSI/status/2066253735332851954
// The original framework in the article can be seen as Wilder-RSI reinterpretation and adaptive normalization scheme,
// but the “sigma” zones should be treated as model-based standardized landmarks, not "guaranteed" empirical probabilities or trading signals.
// Therefore instead of assuming RSI(14) = universal 70/30 behavior, RSI is transformed into logit space;
// its own rolling distribution is estimated then converted to an empirical z-score. Then the calibrated zones are mapped back onto the usual 0–100 RSI scale.
// Newer Version has "Structural Central Line" is added to evaluate the trend. The fast center follows the tactical RSI regime. The slow center provides the structural reference. Residual sigma is estimated independently for more stability. 
// Disclaimer: This indicator is for educational purposes only and is not financial advice. Trading involves risk; use it at your own discretion and responsibility.
//@version=6
indicator("Calibrated RSI Position Zones", shorttitle="C-RSI Zones", overlay=false, precision=2)

//────────────────────────────────────────────────────────────────────
// Helper functions
//────────────────────────────────────────────────────────────────────
f_clamp(x, lo, hi) =>
    math.min(math.max(x, lo), hi)

f_logit_from_rsi(r) =>
    p = f_clamp(r / 100.0, 0.001, 0.999)
    math.log(p / (1.0 - p))

f_rsi_from_logit(x) =>
    100.0 / (1.0 + math.exp(-x))

//────────────────────────────────────────────────────────────────────
// RSI → logit
//────────────────────────────────────────────────────────────────────
rsiRaw   = ta.rsi(src, rsiLen)
logitRsi = f_logit_from_rsi(rsiRaw)
logitSm  = ta.ema(logitRsi, smoothLen)

//────────────────────────────────────────────────────────────────────
// Multi-horizon empirical calibration
//────────────────────────────────────────────────────────────────────
// Fast tactical center and slow structural center.
muFast = ta.ema(logitSm, fastCenterLen)
muSlow = ta.ema(logitSm, slowCenterLen)

// Estimate tactical dispersion from residuals around the SAME EMA
// center used in the numerator. This avoids mixing an EMA center with
// a standard deviation calculated around a different rolling mean.
fastResidual = logitSm - muFast
fastVariance = ta.ema(fastResidual * fastResidual, sigmaLen)
sigmaFast    = math.sqrt(math.max(fastVariance, 0.000001))

// Tactical standardized displacement.
z       = fastResidual / sigmaFast
zSignal = ta.ema(z, 3)
zRank   = ta.percentrank(z, rankLen)

// Structural residual dispersion provides a stable normalization
// reference for regime separation and center slope.
slowResidual = logitSm - muSlow
slowVariance = ta.ema(slowResidual * slowResidual, slowCenterLen)
sigmaSlow    = math.sqrt(math.max(slowVariance, 0.000001))

// Fast equilibrium relative to the structural equilibrium.
regimeZ = (muFast - muSlow) / sigmaSlow

bullishMomentumRegime = regimeZ > regimeThreshold
bearishMomentumRegime = regimeZ < -regimeThreshold
neutralMomentumRegime = not bullishMomentumRegime and not bearishMomentumRegime

momentumRegimeText =
     bullishMomentumRegime ? "Bullish" :
     bearishMomentumRegime ? "Bearish" :
                             "Neutral / Transition"

//────────────────────────────────────────────────────────────────────
// Yellow-center slope
//────────────────────────────────────────────────────────────────────
// Regression slope of muFast in logit space.
// Units: logit-RSI units per bar.
muSlopePerBar =
     ta.linreg(muFast, centerSlopeLen, 0) -
     ta.linreg(muFast, centerSlopeLen, 1)

// Convert the fitted per-bar slope to fitted movement across the
// complete slope window, then normalize by structural sigma.
// Units: structural sigma per slope window.
centerSlopeScoreRaw =
     muSlopePerBar * (centerSlopeLen - 1) / sigmaSlow

centerSlopeScore =
     ta.ema(centerSlopeScoreRaw, centerSlopeSmoothLen)

centerRising  = centerSlopeScore > centerSlopeThreshold
centerFalling = centerSlopeScore < -centerSlopeThreshold
centerFlat    = not centerRising and not centerFalling

centerDirectionText =
     centerRising  ? "Rising" :
     centerFalling ? "Falling" :
                     "Flat"

// Useful secondary diagnostics.
centerSlopeAcceleration = centerSlopeScore - centerSlopeScore[1]
bullishCenterTurn       = ta.crossover(centerSlopeScore, 0.0)
bearishCenterTurn       = ta.crossunder(centerSlopeScore, 0.0)

//────────────────────────────────────────────────────────────────────
// Map calibrated values back to the RSI scale
//────────────────────────────────────────────────────────────────────
centerFast = f_rsi_from_logit(muFast)
centerSlow = f_rsi_from_logit(muSlow)

upper1 = f_rsi_from_logit(muFast + zInner * sigmaFast)
lower1 = f_rsi_from_logit(muFast - zInner * sigmaFast)

upper2 = f_rsi_from_logit(muFast + zOuter * sigmaFast)
lower2 = f_rsi_from_logit(muFast - zOuter * sigmaFast)

upper3 = f_rsi_from_logit(muFast + zExtreme * sigmaFast)
lower3 = f_rsi_from_logit(muFast - zExtreme * sigmaFast)

// Visible yellow-line slope in RSI points per bar.
// This is for human interpretation; centerSlopeScore controls evaluation.
centerRsiSlopePerBar =
     ta.linreg(centerFast, centerSlopeLen, 0) -
     ta.linreg(centerFast, centerSlopeLen, 1)

//────────────────────────────────────────────────────────────────────
// Price-trend and volatility context
//────────────────────────────────────────────────────────────────────
trendEma   = ta.ema(src, trendLen)
trendSlope = trendEma - trendEma[trendSlopeN]

trendState =
     src > trendEma and trendSlope > 0 ? 1 :
     src < trendEma and trendSlope < 0 ? -1 :
                                         0

trendText = trendState == 1  ? "Uptrend" : trendState == -1 ? "Downtrend" : "Range / Transition"

atrPct  = ta.atr(atrLen) / src
volRank = ta.percentrank(atrPct, volRankLen)

volText = volRank >= 80 ? "High Volatility" : volRank <= 20 ? "Low Volatility" : "Normal Volatility"

//────────────────────────────────────────────────────────────────────
// Classified context signals
//────────────────────────────────────────────────────────────────────
statOversold   = z < -zOuter
statOverbought = z > zOuter

extremeHigh = z > zExtreme
extremeLow  = z < -zExtreme

// Re-entry is the tactical event. Empirical percentile confirmation
// verifies that the preceding observation was also in an actual tail.
longReentryRaw  = ta.crossover(z, -zOuter)
shortReentryRaw = ta.crossunder(z, zOuter)

longTailConfirmed =
     not useTailConfirmation or
     (not na(zRank[1]) and zRank[1] <= lowerTailPercentile)

shortTailConfirmed =
     not useTailConfirmation or
     (not na(zRank[1]) and zRank[1] >= upperTailPercentile)

longReentry =
     showSignals and
     longReentryRaw and
     longTailConfirmed and
     zSignal > zSignal[1]

shortReentry =
     showSignals and
     shortReentryRaw and
     shortTailConfirmed and
     zSignal < zSignal[1]

// 1) Continuation: price trend, momentum regime, and center slope agree.
strongLongContinuation =
     longReentry and
     trendState == 1 and
     bullishMomentumRegime and
     centerRising

strongShortContinuation =
     shortReentry and
     trendState == -1 and
     bearishMomentumRegime and
     centerFalling

// 2) Early reversal: structural regime is still adverse, but the fast
// center has already turned in the direction of the re-entry.
earlyLongReversal =
     longReentry and
     not strongLongContinuation and
     bearishMomentumRegime and
     centerRising

earlyShortReversal =
     shortReentry and
     not strongShortContinuation and
     bullishMomentumRegime and
     centerFalling

// 3) Mean reversion: not a continuation or early reversal, but neither
// price context nor center direction is strongly hostile.
meanReversionLong =
     longReentry and
     not strongLongContinuation and
     not earlyLongReversal and
     trendState >= 0 and
     not centerFalling

meanReversionShort =
     shortReentry and
     not strongShortContinuation and
     not earlyShortReversal and
     trendState <= 0 and
     not centerRising

// 4) Countertrend: all remaining valid re-entries.
countertrendLong =
     longReentry and
     not strongLongContinuation and
     not earlyLongReversal and
     not meanReversionLong

countertrendShort =
     shortReentry and
     not strongShortContinuation and
     not earlyShortReversal and
     not meanReversionShort

signalText =
     strongLongContinuation  ? "Strong Long Continuation" :
     strongShortContinuation ? "Strong Short Continuation" :
     earlyLongReversal       ? "Early Bullish Reversal" :
     earlyShortReversal      ? "Early Bearish Reversal" :
     meanReversionLong       ? "Long Mean Reversion" :
     meanReversionShort      ? "Short Mean Reversion" :
     countertrendLong        ? "Countertrend Long Bounce" :
     countertrendShort       ? "Countertrend Short Pullback" :
                               "No new re-entry"

signalTextColor =
     strongLongContinuation  ? color.lime :
     strongShortContinuation ? color.red :
     earlyLongReversal       ? color.aqua :
     earlyShortReversal      ? color.fuchsia :
     meanReversionLong       ? color.teal :
     meanReversionShort      ? color.orange :
     countertrendLong        ? color.silver :
     countertrendShort       ? color.silver :
                               color.white
