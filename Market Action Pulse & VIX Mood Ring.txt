f_dailyChecks(_offset) =>
    c     = close[_offset]
    ema3  = ta.ema(close, 3)[_offset]
    ema5  = ta.ema(close, 5)[_offset]
    ema10 = ta.ema(close, 10)[_offset]
    ema20 = ta.ema(close, 20)[_offset]

    aboveRising5 = c > ema5 and ema5 > ema5[1]
    dailyBuy     = c > ema10 and c > ema20 and ema10 > ema20
    breakattempt = c[1] < ema3[1] and ema3[1] < ema5[1] and c > ema5 and ema3 < ema5 and ema5 < ema10
    strongerbreakattempt = c[1] < c and c > ema5 and ema5 > ema20 and ema20 < ema10

    [c, ema5, ema10, ema20, aboveRising5, dailyBuy, breakattempt, strongerbreakattempt]

[dClose, dSma5, dSma10, dSma20, qqqeAboveRising5, dailyBuySignal, dailyBreakAttempt, dailyStrongerBreakAttempt] =
     request.security(

var float mccEmaFastState       = na
var float mccEmaSlowState       = na
var float mccOscState           = na
var float mccSummationState     = na
var float mccSummationPrevState = na
var float mccMsiEmaState        = na
var float mccMsiEmaPrevState    = na

if mccCalcBar and not na(netAdv)
    oldSummation = mccSummationState
    oldMsiEma    = mccMsiEmaState

    mccEmaFastState := na(mccEmaFastState) ? netAdv : mccEmaFastState + mccAlphaFast * (netAdv - mccEmaFastState)
    mccEmaSlowState := na(mccEmaSlowState) ? netAdv : mccEmaSlowState + mccAlphaSlow * (netAdv - mccEmaSlowState)

    mccOscState := mccEmaFastState - mccEmaSlowState

    mccSummationPrevState := oldSummation
    mccMsiEmaPrevState    := oldMsiEma

    mccSummationState := na(oldSummation) ? mccOscState : oldSummation + mccOscState
    mccMsiEmaState    := na(oldMsiEma) ? mccSummationState : oldMsiEma + msiAlpha * (mccSummationState - oldMsiEma)

mccSummation = mccSummationState
mccOsc       = mccOscState
mccMsiEma    = mccMsiEmaState

mccUptrend =
     not na(mccSummation) and
     not na(mccMsiEma) and
     not na(mccMsiEmaPrevState) and
     mccSummation > mccMsiEma and
     mccMsiEma > mccMsiEmaPrevState

     not na(mccSummation) and
     not na(mccSummationPrevState) and
     mccSummation > mccSummationPrevState and
     mccOsc > 0

f_exposureRule(_state, _prevState) =>
    string rule = ""

    if _state == "GREEN"
        rule := "Build toward full exposure progressively."
    else if _state == "BLUE" and (_prevState == "Orange" or _prevState == "Yellow")
        rule := "Risky→Blue: build up to ~60%; add more if de-risked."
    else if _state == "BLUE" and not (_prevState == "Orange" or _prevState == "Yellow")
        rule := "Blue: add more if positions gain traction."
    else if _state == "YELLOW" and _prevState == "GREEN"
        rule := "Green→Yellow: raise stops, trim heat, protect gains."
    else if _state == "YELLOW"
        rule := "From cash→Yellow: build up to ~30%; add more if de-risked."
    else if _state == "AMBER"
        rule := "Test only 1–2 trades; add more if de-risked."
    else if _state == "ORANGE"
        rule := "No new trades; reduce open heat and protect gains."
    else
        rule := "Exit longs / stay cash; tactical shorts allowed with hard stops."

    rule

exposureRule = f_exposureRule(tmhDashboardState, tmhDashboardState[1])

f_clamp100(_x) =>
    math.max(0.0, math.min(100.0, _x))

f_scoreVix(_x) =>
    float s = na

    if not na(_x)
        if _x < 13.0
            s := 100.0
        else if _x < 17.0
            s := 100.0 - (_x - 13.0) * 15.0 / 4.0
        else if _x < 20.0
            s := 85.0 - (_x - 17.0) * 15.0 / 3.0
        else if _x < 25.0
            s := 70.0 - (_x - 20.0) * 10.0 / 5.0
        else if _x < 32.0
            s := 60.0 - (_x - 25.0) * 40.0 / 7.0
        else
            s := 20.0 - math.min(20.0, (_x - 32.0) * 2.5)

    na(s) ? na : f_clamp100(s)

f_scoreTerm(_r) =>
    float s = na

    if not na(_r)
        if _r < 0.85
            s := 80.0 + math.min(20.0, (0.85 - _r) * 100.0)
        else if _r < 0.95
            s := 80.0 - (_r - 0.85) * 22.0 / 0.10
        else if _r < 1.00
            s := 58.0 - (_r - 0.95) * 28.0 / 0.05
        else
            s := 30.0 - math.min(30.0, (_r - 1.00) * 300.0)

    na(s) ? na : f_clamp100(s)

f_scoreVvix(_x) =>
    float s = na

    if not na(_x)
        if _x < 85.0
            s := 100.0
        else if _x < 100.0
            s := 90.0 - (_x - 85.0) * 20.0 / 15.0
        else if _x < 115.0
            s := 70.0 - (_x - 100.0) * 15.0 / 15.0
        else if _x < 130.0
            s := 55.0 - (_x - 115.0) * 20.0 / 15.0
        else
            s := 35.0 - math.min(35.0, (_x - 130.0) * 2.0)

    na(s) ? na : f_clamp100(s)

f_scoreSkew(_x) =>
    float s = na

    if not na(_x)
        if _x < 115.0
            s := 100.0
        else if _x < 130.0
            s := 82.0 - (_x - 115.0) * 16.0 / 15.0
        else if _x < 145.0
            s := 66.0 - (_x - 130.0) * 25.0 / 15.0
        else
            s := 41.0 - math.min(41.0, (_x - 145.0) * 2.0)

    na(s) ? na : f_clamp100(s)

//──────────────────────────────────────────────────────────────────────────────
// Playbook rules
//──────────────────────────────────────────────────────────────────────────────
f_playbook(_state, _divergence) =>
    string rule = ""

    if _divergence == "Backwardation"
        rule := "Cash 1st, avoid forcing breakouts \n till term structure normalizes."
    else if _divergence == "Complacency Trap"
        rule := "Tighten stops, avoid adding margin, \n demand only A+ setups."
    else if _divergence == "Whipsaw Risk"
        rule := "Keep overnight exposure light, \n prefer intraday entries or smaller starter positions."
    else if _state == "GREEN"
        rule := "Growth-stock environment is supportive \n build exposure progressively."
    else if _state == "BLUE"
        rule := "Constructive tape, buy quality setups, \n avoid chasing extended names."
    else if _state == "YELLOW"
        rule := "Neutral tape, 1/2 positions on new entries; \n wait for follow-through before being aggressive."
    else if _state == "AMBER"
        rule := "Caution! Test only the best setups; \n protect open profits."
    else if _state == "ORANGE"
