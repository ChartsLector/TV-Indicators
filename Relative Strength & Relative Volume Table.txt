// ====================== 2. Static Data Acquisition and Functions ======================

// Helper: Determine conditional color based on RS value
f_get_rs_color_cond(rs_value) =>
    color_to_use = color(na)
    if not na(rs_value)
        if rs_value >= highlight_strong_pct and enable_strong_highlight
            color_to_use := highlight_strong_color
        else if rs_value < highlight_weak_pct and enable_weak_highlight
            color_to_use := highlight_weak_color
        else if rs_value >= highlight_weak_pct and rs_value < highlight_strong_pct and enable_neutral_highlight
            color_to_use := highlight_neutral_color
    color_to_use

// Helper: RVOL text color based on same-symbol daily close direction
f_get_rvol_text_color(_up_flag) =>
    na(_up_flag) ? text_color : (_up_flag >= 1.0 ? rvol_up_color : rvol_down_color)

// Helper: securely retrieve RS + RVOL in one ticker request.
// This avoids the 64 unique request.*() runtime limit when 35 tickers are enabled.
// RS is still calculated as PercentRank of ticker / benchmark.
// RVOL is volume / SMA(volume, rel_vol_len) on the same Daily request used by RS when RVOL is enabled.
f_get_rs_vol_data(_ticker, _tf, _l1, _l2, _l3, _l4, _vlen) =>
    if str.length(_ticker) > 0
        _bench_close = request.security(rs_benchmark, _tf, close, ignore_invalid_symbol=true)
        request.security(_ticker, _tf, [
             ta.percentrank(close / _bench_close, _l1),
             ta.percentrank(close / _bench_close, _l2),
             ta.percentrank(close / _bench_close, _l3),
             ta.percentrank(close / _bench_close, _l4),
             100 * volume / ta.sma(volume, _vlen),
             na(close[1]) ? na : (close >= close[1] ? 1.0 : 0.0)
             ], ignore_invalid_symbol=true)
    else
        [na, na, na, na, na, na]

    array_split = str.split(ticker_name, ':')
    no_prefix = array.size(array_split) > 1 ? array.get(array_split, array.size(array_split) - 1) : ticker_name
    
    // 2. Remove Common Garbage Characters
    s1 = str.replace_all(no_prefix, '"', '')
    s2 = str.replace_all(s1, '}', '')
    s3 = str.replace_all(s2, '{', '')
    s3
